### 2. 实现“确定”按钮功能

实现目标：用户在下拉框选择一个条目，点击“确定”后，在下方显示该条目的详细信息卡片。

#### 2.1. 拓展store

参考：[响应式 API：核心  computed() | Vue.js](https://cn.vuejs.org/api/reactivity-core.html#computed)
```ts
// src/stores/card

const selectedId = ref<string>('')

// 返回选中选项的完整内容
const selectedCard = computed(()=>{
	return cards.value.find((card)=>card.id === selectedId.value) ?? null
})

// setter: 将选中的项的 id 赋值给 selectedId
const setSelected(id: string){
	selectedId.value = id
}

return {selectedId, selectedCard, setSelected}
```

#### 2.2. 编写确认方法
参考：[ECMAScript5.1中文版 + ECMAScript3 + ECMAScript（合集）| some](https://yanhaijing.com/es5/#359)、[JS数组方法](https://zh.javascript.info/array-methods#zai-shu-zu-zhong-sou-suo)
```ts
// 自定义hook
// src/composables/homeCard.ts
import { ref } from 'vue'
import { useCardStore } from '../stores/card'
import { storeToRefs } from 'pinia'
import type { CardItem } from '../types/card'

export function useHomeConfirm(){
	const cardstore = useCardStore()
	const { cards } = storeToRefs(cardstore)
	
	// 创建一个用于存储下拉框被选中的变量，该变量对应的是卡片的id
	const tempSelectedId = ref<string>('')
	// 创建存储被选中的卡片的数组
	const selectedCardList = ref<CardItem[]>([])
	
	// 编写确认按钮的方法
	const confirm = () => {
		// 确认 store 中是否有该卡片
		cardStore.setSelected(tempSelectedId.value)
		if(cardStore.selectedCard == null){
			alert("请选择一个有效的卡片")
			return
		}
		
		// 检查该项是否已经被导入到数组中
		const exist = selectCardList.value.some((item) => {item.id === tempSelectedId.value})
		if(exist){
			alert("该卡片已存在")
			return
		}
		
		// 控制数组种得卡片数量
		if(selectedCardList.value.length >= 10){
			alert("最多只能选十张卡片")
			return
		}
		
		// 插入卡片并清空 tempSelectedId
		selectedCardList.value.push(cardStore.selectedCard)
		tempSelectedId = ''
	}
	
	// 编写删除卡片的方法
	const removeCard = (id: string) => {
		try{
			selectedCardList.value.splice(
			selectedCardList.value.findIndex((item) => item.id === id), 1
		}catch (error) {
			console.error("未找到该卡片:", error);
		}
		alert(`删除成功,ID${id}，${selectedCardList.value.length}张卡片剩余`)
	}
	
	return{
		cards,
		tempSelectedId,
		selectedCardList,
		confirm,
		removeCard
	}
}
```

> 这里有一点需要提醒，为什么之前下拉框默认选项 selected 的选项不见了？在 Vue 中使用 `v-model` 绑定 `<select>` 时，所有 `option` 上的 `selected` 属性都会被忽略。默认选中项完全由绑定的数据（如 `tempSelectedId`）的初始值决定。如果初始值为空，并且没有 `value` 匹配的 `option`（包括空字符串匹配），那么下拉框会显示空白或第一个未禁用的 `option`（具体行为依赖浏览器）。因此，手动添加的 `selected` 属性会失效。即响应式的优先级高于原生属性。

### 3. 实现新增功能

实现目标：点击“新增”按钮，左侧滑出表单面板，填写数据后通过 axios 发送请求（由 mockjs 拦截），将数据存入 Pinia，下拉框自动出现新选项并选中。
使用mock的目的：从 store（按照正常项目是从服务器获取，而我们这是通过mock拦截这个获取，转到从本地）fetch数据到页面最终显示在下拉框中 ；从表单提交到服务器时，通过mock拦截并传输到本地 store 中
#### 3.1. 配置mock

参考： [mock | Better-Mock](https://lavyun.github.io/better-mock/document/mock/#mock)、[JSON 方法，toJSON](https://zh.javascript.info/json#jsonparse)、[TypeScript 中文网: 文档 - 工具类型 omit](https://ts.nodejs.cn/docs/handbook/utility-types.html#omittype-keys)、[TypeScript 中文网: 文档 - 日常类型 - 类型断言 as](https://ts.nodejs.cn/docs/handbook/2/everyday-types.html#%E7%B1%BB%E5%9E%8B%E6%96%AD%E8%A8%80)、[对象的扩展 - ECMAScript 6入门](https://es6.ruanyifeng.com/#docs/object#%E5%AF%B9%E8%B1%A1%E7%9A%84%E6%89%A9%E5%B1%95%E8%BF%90%E7%AE%97%E7%AC%A6)
```ts
// src/mock/index.ts
import mock from 'better-mock'
import type { CardItem } from '../types/card'

// 用内存数组模拟数据库（也可后续持久化）
let db: CardItem[] = [
	{ id: "1", name: "阅读", time: "2026-05-13", description: "阅读 Vue 文档" },
	{ id: "2", name: "运动", time: "2026-05-14", description: "跑步 30 分钟" },
	{ id: "3", name: "购物", description: "买一些日用品" },
]

// 获取所有条目（后续可持续优化 store）
mock.mock('/api/cards', 'get', () => db)

mock.mock('/api/cards', 'post', (options: { body: string }) => {
	const body = JSON.parse(options.body) as omit<CardItem, 'id'>
	const newCard: CardItem = {
		// 随机生成id标识，参考 mock.Random 部分
		id: mock.Random.guid(),
		// 对象拓展运算符，将表单填写的内容和生成的 id 组合成为
		...body
	}
	bd.push(newCard)
	return newCard
})
```

#### 3.2. 封装 api 请求

参考：[Axios 实例 | Axios中文文档 | Axios中文网](https://www.axios-http.cn/docs/instance)
```ts
// src/apis/card.ts
import axios from 'axios'
import type { CardItem } from '../types/card'

// 创建 Axios 实例, baseURL 为空字符串，意味着请求会以相对路径发送
const http = axios.create({baseURL: ''})

// 导出一个包含 fetch 和 add 两个方法的对象
export const itemApi = {

	addItem:(data:Omit<CardItem, 'id'>) => http.post<CardItem>('/api/cards', data),
}
```

#### 3.3.  改造 Store：从 Mock 初始化数据，异步添加

参考：[Promise，async/await](https://zh.javascript.info/async)
```ts
// src/stores/card
// src/stores/card.ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import type { CardItem } from '../types/card'

export const useCardStore = defineStore('card', () =>{
	const cards = ref<CardItem[]>([])
	const selectedId = ref<string>('')
	const loading = ref(false)
	
	// 从 store 获取卡片列表
	async function fetchCards() {
		loading.value = true
		try{
			const {data} = await itemApi.fetchItems()
			card.value = data
		}catch(e) {
			console.error("获取数据失败:", e);
		}finally{
			loading.value = false
		}
	} 
	
	// 将卡片添加到 store 中
	async function addCard(card: omit<CardItem, 'id'>) {
		loading.value = true
		try{
			const { data } = await itemApi.addItem(card)
			cards.value.push(data)
			selectedId.value = data.id
		}catch(e) {
			console.error("添加卡片失败:", e);
		}finally{
			loading.value = false
		}
	}
	
	// 返回选中选项的完整内容
	const selectedCard = computed(()=>{
		return cards.value.find((card)=>card.id === selectedId.value) ?? null
	})
	
	// setter: 将选中的项的 id 赋值给 selectedId
	const setSelected(id: string){
		selectedId.value = id
	}
	
	return { 
		cards,
		selectedId, 
		loading, 
		fetchCards, 
		addCard, 
		selectedCard, 
		setSelected 
	}
})
```

```vue
	import { useCardStore } from "../stores/card";
	
	const store = useCardStore();
	const form = ref({
		name: "",
		date1: "",
		date2: "",
		desc: "",
	})

	async function onSubmit() {

			}
		} catch (error) {

			submitting.value = false
```
